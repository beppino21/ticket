# MAIL_DRY_RUN=false -> invia davvero tramite SMTP
MAIL_DRY_RUN=true

# --- SAP OData (S/4HC Public Edition, /TKM/CC_WEB_SRV) ---
# Se queste chiavi mancano, SAPODataConfig usa i default cablati nel
# codice (gli ultimi valori noti funzionanti) — l'app non si rompe,
# ma a quel punto la configurazione reale è quella nel .java, non qui.
SAP_BASE_URL=http://newton.domain.eonegroup.it:8001

# URL pubblico con cui il server ticket è raggiungibile dai browser degli
# utenti (protocollo+host+porta+context path, SENZA slash finale). Usato per
# costruire il link diretto al ticket nelle email di notifica.
# Se non valorizzato, il link viene omesso dall'email (nessun errore bloccante).
APP_BASE_URL=https://ticket.lamplast.it:50000/ticket

# Sollecito automatico chiusura ticket — mail giornaliera aggregata al
# backoffice per i ticket fermi da troppi giorni in "Richiesta chiusura
# ticket" o "Ticket risolto" (impostati dal cliente), non ancora chiusi sul
# backend SAP. Nessuna schedulazione esterna necessaria: parte da sola
# all'avvio della webapp (vedi SollecitoChiusuraScheduler) e gira ogni 24h.
SOLLECITO_CHIUSURA_EMAIL=helpdesk@eonegroup.it
SOLLECITO_CHIUSURA_GIORNI=1
SOLLECITO_CHIUSURA_ORA=8
# SOLLECITO_CHIUSURA_ABILITATO=false  -> disattiva del tutto lo scheduler
# NOTA: questo sollecito è attualmente MORTO CODE — nessuno lo invoca (vedi
# PromemoriaSchedulerListener per un esempio di scheduler realmente attivo).

# Promemoria quotidiano ticket pendenti — mail personalizzata a ogni AMS con
# i propri ticket non chiusi (evidenziati a colori per ritardo), e mail
# identica a ogni DISPATCHER con DRAFT + richieste di riattribuzione
# pendenti. Parte da solo all'avvio della webapp (vedi
# PromemoriaSchedulerListener), solo lun-ven, nessuna mail se l'utente non
# ha nulla di pendente.
PROMEMORIA_QUOTIDIANO_ABILITATO=true
PROMEMORIA_QUOTIDIANO_ORA=8
PROMEMORIA_QUOTIDIANO_MINUTO=30

SAP_CLIENT=300
SAP_USER=EONE
SAP_PASS=changeme
